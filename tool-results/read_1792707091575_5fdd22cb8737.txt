    20→/**
    21→ * WE Pairing Server — acts as the "PC" side of the WE pairing protocol.
    22→ *
    23→ * This is the heart of the new WepWawet architecture. Instead of emulating
    24→ * Wine+Box64+Steam+WE, this server NATIVELY speaks the WE pairing protocol.
    25→ * The official WE Android app discovers and pairs with this server exactly
    26→ * as it would pair with a real PC running Wallpaper Engine.
    27→ *
    28→ * Protocol (reverse-engineered from the WE Android APK decompiled with jadx):
    29→ *
    30→ * 1. DISCOVERY (UDP 7884):
    31→ *    - Server broadcasts "WallpaperEngine/Ping:<pid>:<hostname>:1:<ifaceId>"
    32→ *      via UDP multicast to 239.100.0.1:7884 every 5 seconds
    33→ *    - Phone's DiscoverService listens and finds the server
    34→ *
    35→ * 2. PAIRING (TCP 7889):
    36→ *    - Phone connects to server on TCP 7889
    37→ *    - Server sends RSA public key (exponent + modulus as BigInteger)
    38→ *    - Phone generates AES key + IV
    39→ *    - Phone encrypts AES key with server's RSA public key → sends to server
    40→ *    - Phone encrypts EncAuthRequest (JSON) with AES/CBC/PKCS5Padding → sends to server
    41→ *    - Server decrypts AES key with RSA private key
    42→ *    - Server decrypts EncAuthRequest with AES
    43→ *    - Server verifies PIN
    44→ *    - Server encrypts EncAuthResponse (JSON) with AES → sends to phone
    45→ *    - All subsequent communication is AES-encrypted
    46→ *
    47→ * 3. FILE TRANSFER:
    48→ *    - Phone requests wallpaper download
    49→ *    - Server sends .mpkg file via AES-encrypted ranged transfer
    50→ *    - Uses writeInt(length) + ciphertext format for all messages
    51→ *
    52→ * Wire format for encrypted messages:
    53→ *   writeInt(ciphertext_length)   // 4 bytes, LITTLE-ENDIAN (verified from decompiled
    54→ *                                 // TcpClient$Companion.writeInt — output.writeByte(value & 255)
    55→ *                                 // then >> 8, >> 16, >> 24 — that's little-endian!)
    56→ *   write(ciphertext)             // AES/CBC/PKCS5Padding encrypted JSON
    57→ *
    58→ * CRITICAL: The WE Android app's TcpClient$Companion.readInt/writeInt use
    59→ * LITTLE-ENDIAN byte order (ByteBuffer.order(LITTLE_ENDIAN) and shift-down
    60→ * writeByte calls). Java's DataInputStream.readInt() and DataOutputStream.writeInt()
    61→ * use BIG-ENDIAN. So we MUST use our own readLeInt()/writeLeInt() helpers
    62→ * for ALL length prefixes, otherwise the server misreads every message length
    63→ * and pairing fails silently.
    64→ *
    65→ * Heartbeat: After pairing succeeds, the phone's connector thread sends
    66→ * "heartbeat\n" (10 ASCII bytes, no length prefix) every ~3 seconds.
    67→ * The server must detect and skip these frames, otherwise readLeInt() would
    68→ * interpret 'h','e','a','r' as a 4-byte length and bail out.
    69→ *
    70→ * Constants from decompiled TcpClient.java:
    71→ *   MAX_SAFE_COMMAND_LENGTH = 4096
    72→ *   MAX_SAFE_DATA_LENGTH = 1048572  (1 MB)
    73→ */

   113→    /**
   114→     * Generate an RSA keypair for pairing.
   115→     * The public key will be sent to the phone during pairing.
   116→     * The private key is used to decrypt the phone's AES key.
   117→     */
   118→    fun generateRsaKeyPair(): KeyPair {
   119→        WepWawetLogger.proxy("Generating RSA keypair ($RSA_KEY_SIZE bits)")
   120→        val generator = KeyPairGenerator.getInstance("RSA")
   121→        generator.initialize(RSA_KEY_SIZE)
   122→        val keyPair = generator.generateKeyPair()
   123→        rsaKeyPair = keyPair
   124→
   125→        // Log the public key components (for debugging)
   126→        val pubKey = keyPair.public as java.security.interfaces.RSAPublicKey
   127→        WepWawetLogger.proxyDebug("RSA keypair generated",
   128→            "modulus_bits" to pubKey.modulus.bitLength(),
   129→            "exponent" to pubKey.publicExponent.toString(16).take(16))
   130→
   131→        return keyPair
   132→    }

   204→    /**
   205→     * Write a 4-byte LITTLE-ENDIAN integer to the output stream.
   206→     *
   207→     * Mirrors the WE Android app's TcpClient$Companion.writeInt exactly:
   208→     *   output.writeByte(value & 255)
   209→     *   output.writeByte((value >> 8) & 255)
   210→     *   output.writeByte((value >> 16) & 255)
   211→     *   output.writeByte((value >> 24) & 255)
   212→     *
   213→     * If you use DataOutputStream.writeInt() instead, that uses BIG-ENDIAN,
   214→     * and the phone will misread every length prefix — pairing will fail
   215→     * silently with a 0 or huge bogus length.
   216→     */
   217→    private fun writeLeInt(output: DataOutputStream, value: Int) {
   218→        output.writeByte(value and 0xFF)
   219→        output.writeByte((value shr 8) and 0xFF)
   220→        output.writeByte((value shr 16) and 0xFF)
   221→        output.writeByte((value shr 24) and 0xFF)
   222→    }
   223→
   224→    /**
   225→     * Strip leading zero bytes from a BigInteger.toByteArray() result.
   226→     * The real protocol sends the modulus as raw bytes WITHOUT the sign byte
   227→     * that Java's BigInteger.toByteArray() prepends for positive numbers with
   228→     * the high bit set.
   229→     */
   230→    private fun stripLeadingZero(bytes: ByteArray): ByteArray {
   231→        var start = 0
   232→        while (start < bytes.size - 1 && bytes[start] == 0.toByte()) {
   233→            start++
   234→        }
   235→        return bytes.copyOfRange(start, bytes.size)
   236→    }

   238→    /**
   239→     * Read a length-prefixed message: 4-byte LE length + N bytes.
   240→     * Returns null on invalid length (caller should close the connection).
   241→     */
   242→    private fun readLengthPrefixed(input: DataInputStream, label: String): ByteArray? {
   243→        val length = readLeInt(input)
   244→        WepWawetLogger.proxy("Reading length-prefixed: $label", "length" to length)
   245→        if (length <= 0 || length > MAX_SAFE_DATA_LENGTH) {
   246→            WepWawetLogger.proxyError("Invalid $label length: $length")
   247→            return null
   248→        }
   249→        val data = ByteArray(length)
   250→        input.readFully(data)
   251→        return data
   252→    }
   253→
   254→    /**
   255→     * Read a 4-byte LITTLE-ENDIAN integer from the input stream.
   256→     *
   257→     * Mirrors the WE Android app's TcpClient$Companion.readInt exactly:
   258→     *   byte[] bArr = new byte[4];
   259→     *   input.read(bArr, 0, 4);
   260→     *   return ByteBuffer.wrap(bArr).order(LITTLE_ENDIAN).getInt();
   261→     *
   262→     * If you use DataInputStream.readInt() instead, that uses BIG-ENDIAN,
   263→     * and you'll misread every length prefix from the phone.
   264→     */
   265→    private fun readLeInt(input: DataInputStream): Int {
   266→        val b = ByteArray(4)
   267→        input.readFully(b)
   268→        return (b[0].toInt() and 0xFF) or
   269→               ((b[1].toInt() and 0xFF) shl 8) or
   270→               ((b[2].toInt() and 0xFF) shl 16) or
   271→               ((b[3].toInt() and 0xFF) shl 24)
   272→    }
   273→
   274→    /**
   275→     * Read a length-prefixed message, gracefully skipping 'heartbeat\n'
   276→     * keepalive frames that the phone's connector thread sends every ~3s
   277→     * after pairing. Without this, readLeInt() would read 'h','e','a','r'
   278→     * (0x68, 0x65, 0x61, 0x72) → little-endian 0x72616568 = 1,919,014,504,
   279→     * which exceeds MAX_SAFE_DATA_LENGTH and breaks the command loop.
   280→     *
   281→     * Strategy: peek the first 4 bytes. If they spell ASCII "hear",
   282→     * consume the remaining 6 bytes ("tbeat\n") and return a sentinel
   283→     * indicating a heartbeat was seen.
   284→     */
   285→    private fun readMessageOrHeartbeat(input: DataInputStream): Pair<String, ByteArray?> {
   286→        val header = ByteArray(4)
   287→        input.readFully(header)
   288→        // Check for 'h','e','a','r' = first 4 bytes of "heartbeat\n"
   289→        if (header[0] == 0x68.toByte() && header[1] == 0x65.toByte() &&
   290→            header[2] == 0x61.toByte() && header[3] == 0x72.toByte()) {
   291→            // Consume the remaining 6 bytes of "heartbeat\n": t,b,e,a,t,\n
   292→            val rest = ByteArray(6)
   293→            input.readFully(rest)
   294→            return "heartbeat" to null
   295→        }
   296→        // Otherwise treat as little-endian length prefix
   297→        val length = (header[0].toInt() and 0xFF) or
   298→                     ((header[1].toInt() and 0xFF) shl 8) or
   299→                     ((header[2].toInt() and 0xFF) shl 16) or
   300→                     ((header[3].toInt() and 0xFF) shl 24)
   301→        if (length <= 0 || length > MAX_SAFE_DATA_LENGTH) {
   302→            return "invalid" to null
   303→        }
   304→        val payload = ByteArray(length)
   305→        input.readFully(payload)
   306→        return "message" to payload
   307→    }

   309→    /**
   310→     * Handle a single pairing connection from the WE Android app.
   311→     *
   312→     * This implements the full pairing handshake:
   313→     * 1. Send RSA public key (modulus + exponent, length-prefixed LE)
   314→     * 2. Receive encrypted AES key (RSA-encrypted, length-prefixed LE)
   315→     * 3. Receive AES IV (16 raw bytes, length-prefixed LE)
   316→     * 4. Decrypt AES key with RSA private key (RSA/ECB/OAEPPadding)
   317→     *    VERIFIED from decompiled TcpClient line 541
   318→     * 5. Receive encrypted EncAuthRequest (AES/CBC/PKCS5Padding)
   319→     * 6. Decrypt with AES, parse JSON, verify PIN
   320→     * 7. Send encrypted EncAuthResponse (AES/CBC/PKCS5Padding)
   321→     * 8. Enter command loop (skips heartbeat frames)
   322→     */
   323→    private fun handleConnection(client: Socket) {
   324→        var pairingSuccess = false
   325→        try {
   326→            client.use {
   327→                // Set TCP keepalive + timeouts
   328→                it.keepAlive = true
   329→                it.soTimeout = 60000  // 60s timeout (phone blocks on commandQueue until PIN entered)
   330→
   331→                // CRITICAL: Use BufferedOutputStream to avoid TCP fragmentation.
   332→                // Without buffering, writeLeInt sends 4 individual bytes via writeByte,
   333→                // creating 4 tiny TCP segments. The phone may time out waiting for
   334→                // the complete length prefix.
   335→                val input = DataInputStream(java.io.BufferedInputStream(it.getInputStream()))
   336→                val output = DataOutputStream(java.io.BufferedOutputStream(it.getOutputStream()))
   337→
   338→                WepWawetLogger.proxy("Handling pairing connection",
   339→                    "remote" to it.remoteSocketAddress.toString())
   340→
   341→                // === PC → phone: Send RSA exponent FIRST (as 3-byte BE BigInteger) ===
   342→                // VERIFIED from real capture: Frame 0 = length=3 + bytes `01 00 01` = 65537
   343→                // (was modulus-first in earlier code — WRONG)
   344→                val (modulus, exponent) = getRsaPublicKeyComponents()
   345→                val expBytes = stripLeadingZero(exponent.toByteArray())
   346→                WepWawetLogger.proxy("Sending RSA exponent",
   347→                    "exponent" to exponent.toString(16),
   348→                    "wire_bytes" to expBytes.size,
   349→                    "wire_hex" to expBytes.joinToString("") { "%02x".format(it) })
   350→                writeLeInt(output, expBytes.size)
   351→                output.write(expBytes)
   352→                output.flush()
   353→
   354→                // === PC → phone: Send RSA modulus (128 raw BE bytes for RSA-1024) ===
   355→                // VERIFIED: 128 bytes, no sign byte (BigInteger.toByteArray may include
   356→                // a leading 0x00 sign byte — strip it).
   357→                val modBytes = stripLeadingZero(modulus.toByteArray())
   358→                // Pad to exactly 128 bytes if needed (left-pad with zeros)
   359→                val modPadded = if (modBytes.size < RSA_BLOCK_SIZE) {
   360→                    ByteArray(RSA_BLOCK_SIZE - modBytes.size) + modBytes
   361→                } else {
   362→                    modBytes
   363→                }
   364→                WepWawetLogger.proxy("Sending RSA modulus",
   365→                    "modulus_bits" to modulus.bitLength(),
   366→                    "wire_bytes" to modPadded.size,
   367→                    "wire_hex_first_16" to modPadded.take(16).joinToString("") { "%02x".format(it) })
   368→                writeLeInt(output, modPadded.size)
   369→                output.write(modPadded)
   370→                output.flush()
   371→
   372→                // === Phone → PC: Receive 3 RSA-encrypted frames (128B each) ===
   373→                // VERIFIED from decompiled TcpClient connector$lambda$2:
   374→                //   Frame 0: RSA-OAEP-encrypted deviceUUID (line 548-558)
   375→                //   Frame 1: RSA-OAEP-encrypted AES IV (line 578-586)
   376→                //   Frame 2: RSA-OAEP-encrypted AES key (line 588-596)
   377→                // The phone sends UUID FIRST, then IV, then AES key.
   378→                // (Our previous code read them as AES key → IV → third — WRONG!)
   379→                val encDeviceUUID = readLengthPrefixed(input, "RSA-encrypted deviceUUID")
   380→                    ?: return
   381→                val encIv = readLengthPrefixed(input, "RSA-encrypted AES IV")
   382→                    ?: return
   383→                val encAesKey = readLengthPrefixed(input, "RSA-encrypted AES key")
   384→                    ?: return
   385→
   386→                // Decrypt with RSA/ECB/OAEPPadding — VERIFIED from decompiled
   387→                // TcpClient.connector$lambda$2 line 541:
   388→                //   Cipher.getInstance("RSA/ECB/OAEPPadding")
   389→                // NOT PKCS1Padding — using the wrong padding = pairing failure!
   390→                val rsaCipher = Cipher.getInstance("RSA/ECB/OAEPPadding")
   391→                rsaCipher.init(Cipher.DECRYPT_MODE, rsaKeyPair!!.private)
   392→                val deviceUUIDBytes = rsaCipher.doFinal(encDeviceUUID)
   393→                val ivBytes = rsaCipher.doFinal(encIv)
   394→                val aesKeyBytes = rsaCipher.doFinal(encAesKey)
   395→
   396→                currentAesKey = SecretKeySpec(aesKeyBytes, "AES")
   397→                currentIv = IvParameterSpec(ivBytes)
   398→
   399→                WepWawetLogger.proxy("AES key exchange complete",
   400→                    "aes_key_bits" to (aesKeyBytes.size * 8),
   401→                    "iv_bytes" to ivBytes.size,
   402→                    "device_uuid" to String(deviceUUIDBytes, Charsets.UTF_8),
   403→                    "device_uuid_bytes" to deviceUUIDBytes.size)
   404→
   405→                // Step 4: Receive encrypted EncAuthRequest (length LITTLE-ENDIAN!)
   406→                val aesCipher = Cipher.getInstance("AES/CBC/PKCS5Padding")
   407→                aesCipher.init(Cipher.DECRYPT_MODE, currentAesKey, currentIv)
   408→
   409→                val encAuthReqLen = readLeInt(input)
   410→                WepWawetLogger.proxyDebug("Received encrypted EncAuthRequest",
   411→                    "length" to encAuthReqLen)
   412→                if (encAuthReqLen <= 0 || encAuthReqLen > MAX_SAFE_DATA_LENGTH) {
   413→                    WepWawetLogger.proxyError("Invalid EncAuthRequest length: $encAuthReqLen")
   414→                    return
   415→                }
   416→                val encAuthReqBytes = ByteArray(encAuthReqLen)
   417→                input.readFully(encAuthReqBytes)
   418→
   419→                val authReqJson = String(aesCipher.doFinal(encAuthReqBytes), Charsets.UTF_8)
   420→                WepWawetLogger.proxy("Received EncAuthRequest",
   421→                    "json" to authReqJson)
   422→
   423→                // Step 5: Parse EncAuthRequest (JSON via Gson)
   424→                // Expected fields: pin, name, deviceUserName, displayWidth, displayHeight,
   425→                    // version, mpkgVersion, appVersion
   426→                val gson = com.google.gson.Gson()
   427→                val authReq = gson.fromJson(authReqJson, EncAuthRequest::class.java)
   428→
   429→                WepWawetLogger.proxy("EncAuthRequest parsed",
   430→                    "pin" to (authReq.pin ?: ""),
   431→                    "name" to (authReq.name ?: ""),
   432→                    "deviceUserName" to (authReq.deviceUserName ?: ""),
   433→                    "displayWidth" to (authReq.displayWidth ?: 0),
   434→                    "displayHeight" to (authReq.displayHeight ?: 0),
   435→                    "version" to (authReq.version ?: 0),
   436→                    "mpkgVersion" to (authReq.mpkgVersion ?: ""),
   437→                    "appVersion" to (authReq.appVersion ?: ""))
   438→
   439→                // Step 6: Verify PIN
   440→                val pinValid = authReq.pin == currentPin
   441→                WepWawetLogger.proxy("PIN verification",
   442→                    "received_pin" to (authReq.pin ?: ""),
   443→                    "expected_pin" to currentPin,
   444→                    "valid" to pinValid)
   445→
   446→                // Step 7: Send encrypted EncAuthResponse
   447→                val authResponse = EncAuthResponse()
   448→                // VERIFIED from decompiled TcpClient line 677:
   449→                //   if (encAuthResponse.getStatus().equals("AuthOK"))
   450→                // The success status is "AuthOK", NOT "ok"!
   451→                authResponse.status = if (pinValid) "AuthOK" else "AuthFailed"
   452→
   453→                val respJson = gson.toJson(authResponse)
   454→                WepWawetLogger.proxyDebug("Sending EncAuthResponse",
   455→                    "json" to respJson,
   456→                    "status" to authResponse.status)
   457→
   458→                // Re-init cipher for encryption (AES/CBC is stateful)
   459→                val encCipher = Cipher.getInstance("AES/CBC/PKCS5Padding")
   460→                encCipher.init(Cipher.ENCRYPT_MODE, currentAesKey, currentIv)
   461→                val respBytes = encCipher.doFinal(respJson.toByteArray(Charsets.UTF_8))
   462→                // LITTLE-ENDIAN length prefix!
   463→                writeLeInt(output, respBytes.size)
   464→                output.write(respBytes)
   465→                output.flush()
   466→
   467→                if (!pinValid) {
   468→                    WepWawetLogger.proxyWarn("Pairing failed — invalid PIN")
   469→                    onPairingFailure?.invoke("Invalid PIN")
   470→                    return
   471→                }
   472→
   473→                // Step 8: Pairing successful!
   474→                pairingSuccess = true
   475→                currentClient = client
   476→                it.soTimeout = 0  // Remove timeout for command loop
   477→
   478→                val deviceName = authReq.deviceUserName ?: authReq.name ?: "Unknown"
   479→                WepWawetLogger.proxy("Pairing successful!",
   480→                    "device_name" to deviceName,
   481→                    "app_version" to (authReq.appVersion ?: ""),
   482→                    "mpkg_version" to (authReq.mpkgVersion ?: ""),
   483→                    "display" to "${authReq.displayWidth}x${authReq.displayHeight}")
   484→                onPairingSuccess?.invoke(deviceName)
   485→
   486→                // Step 9: Enter command loop (handle subsequent commands)
   487→                commandLoop(input, output)
   488→
   489→            }
   490→        } catch (e: Exception) {
   491→            WepWawetLogger.proxyError("Pairing connection error",
   492→                "exception" to e.javaClass.name,
   493→                "message" to (e.message ?: ""),
   494→                exception = e)
   495→            if (!pairingSuccess) {
   496→                onPairingFailure?.invoke(e.message ?: "Connection error")
   497→            }
   498→        }
   499→    }

   563→        output: DataOutputStream,
   564→        mpkgFile: java.io.File,
